cmake_minimum_required(VERSION 3.22)
project(ToneScope VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(TONESCOPE_DESKTOP "Build native desktop app" ON)
option(TONESCOPE_CORE_ONLY "Build portable DSP/streaming tests without JUCE or GUI" OFF)
include(CTest)
if(BUILD_TESTING)
  add_executable(probe_tests tests/native/probe_tests.cpp)
  target_include_directories(probe_tests PRIVATE apps/desktop)
  if(NOT WIN32)
    find_package(Threads REQUIRED)
    target_link_libraries(probe_tests PRIVATE Threads::Threads)
  endif()
  add_test(NAME callback_probe COMMAND probe_tests)
endif()
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
endif()
add_library(tonescope_core
  libs/audio-engine/src/Engine.cpp
  libs/audio-engine/src/Wav.cpp
  libs/audio-engine/src/Streaming.cpp
  libs/analysis/src/Analysis.cpp)
target_include_directories(tonescope_core PUBLIC libs/audio-engine/include libs/session/include PRIVATE dsp/generated)
if(TONESCOPE_CORE_ONLY)
  add_executable(core_tests tests/native/core_tests.cpp)
  target_include_directories(core_tests PRIVATE dsp/generated)
  target_link_libraries(core_tests PRIVATE tonescope_core)
  add_test(NAME core COMMAND core_tests)
  add_executable(streaming_tests tests/native/streaming_tests.cpp)
  target_link_libraries(streaming_tests PRIVATE tonescope_core)
  add_test(NAME streaming COMMAND streaming_tests)
  return()
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/.deps/JUCE/CMakeLists.txt")
  add_subdirectory(.deps/JUCE EXCLUDE_FROM_ALL)
else()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 3fb5cbc0340758596dac39f7c8de47d562b4896e)
  FetchContent_MakeAvailable(JUCE)
endif()
juce_add_console_app(tonescope_cli PRODUCT_NAME "ToneScope Renderer")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(TONESCOPE_IDENTITY_HEADER "${CMAKE_BINARY_DIR}/generated/ToneScopeBuildIdentity.h")
add_custom_target(tonescope_build_identity
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/build-identity.py"
    --root "${CMAKE_SOURCE_DIR}" --out "${TONESCOPE_IDENTITY_HEADER}"
    --compiler "${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}"
    --generator "${CMAKE_GENERATOR} ${CMAKE_GENERATOR_PLATFORM}"
  BYPRODUCTS "${TONESCOPE_IDENTITY_HEADER}"
  VERBATIM)
add_dependencies(tonescope_cli tonescope_build_identity)
target_include_directories(tonescope_cli PRIVATE "${CMAKE_BINARY_DIR}/generated")
target_sources(tonescope_cli PRIVATE tools/render-cli/Main.cpp)
target_link_libraries(tonescope_cli PRIVATE tonescope_core juce::juce_core juce::juce_cryptography)
target_compile_definitions(tonescope_cli PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
if(TONESCOPE_DESKTOP)
  set(JUCE_WEBVIEW2_PACKAGE_LOCATION "${CMAKE_SOURCE_DIR}/.deps")
  if(NOT EXISTS "${JUCE_WEBVIEW2_PACKAGE_LOCATION}/Microsoft.Web.WebView2.1.0.3485.44/build/native/include/WebView2.h")
    include(FetchContent)
    FetchContent_Declare(tonescope_webview2
      URL https://api.nuget.org/v3-flatcontainer/microsoft.web.webview2/1.0.3485.44/microsoft.web.webview2.1.0.3485.44.nupkg
      URL_HASH SHA256=bc09150b179246ac90189649b13be8e6b11b3ac200e817e18df106e1f3cf489e
      DOWNLOAD_NAME webview2.zip
      SOURCE_DIR "${JUCE_WEBVIEW2_PACKAGE_LOCATION}/Microsoft.Web.WebView2.1.0.3485.44")
    FetchContent_MakeAvailable(tonescope_webview2)
  endif()
  # JUCE's finder otherwise chooses the first wildcard match or a stale cache.
  set(WebView2_root_dir "${JUCE_WEBVIEW2_PACKAGE_LOCATION}/Microsoft.Web.WebView2.1.0.3485.44" CACHE PATH "Pinned WebView2 SDK" FORCE)
  juce_add_gui_app(ToneScope PRODUCT_NAME "ToneScope" COMPANY_NAME "ToneScope" VERSION ${PROJECT_VERSION} NEEDS_WEBVIEW2 TRUE)
  target_sources(ToneScope PRIVATE apps/desktop/Main.cpp)
  target_include_directories(ToneScope PRIVATE dsp/generated)
  target_link_libraries(ToneScope PRIVATE tonescope_core juce::juce_audio_utils juce::juce_dsp juce::juce_cryptography)
  target_compile_definitions(ToneScope PRIVATE JUCE_USE_CURL=0 JUCE_APPLICATION_NAME_STRING="ToneScope" JUCE_APPLICATION_VERSION_STRING="0.1.0")
  include(apps/desktop/Frontend.cmake)
endif()
if(BUILD_TESTING)
  add_executable(core_tests tests/native/core_tests.cpp)
  target_include_directories(core_tests PRIVATE dsp/generated)
  target_link_libraries(core_tests PRIVATE tonescope_core)
  add_test(NAME core COMMAND core_tests)
  add_executable(streaming_tests tests/native/streaming_tests.cpp)
  target_link_libraries(streaming_tests PRIVATE tonescope_core)
  add_test(NAME streaming COMMAND streaming_tests)
endif()
if(BUILD_TESTING AND NOT TONESCOPE_CORE_ONLY)
  add_executable(transport_tests tests/native/transport_tests.cpp)
  target_include_directories(transport_tests PRIVATE apps/desktop dsp/generated)
  target_link_libraries(transport_tests PRIVATE tonescope_core juce::juce_core juce::juce_cryptography)
  target_compile_definitions(transport_tests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  add_test(NAME transport COMMAND transport_tests)
  add_executable(assistant_bridge_tests tests/native/assistant_bridge_tests.cpp)
  target_link_libraries(assistant_bridge_tests PRIVATE tonescope_core juce::juce_core juce::juce_cryptography)
  target_compile_definitions(assistant_bridge_tests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  find_program(TONESCOPE_NODE_EXECUTABLE NAMES node.exe node REQUIRED)
  add_dependencies(assistant_bridge_tests tonescope_cli)
  add_test(NAME assistant_bridge COMMAND assistant_bridge_tests)
  set_tests_properties(assistant_bridge PROPERTIES TIMEOUT 90 ENVIRONMENT
    "TONESCOPE_ASSISTANT_DIR=${CMAKE_SOURCE_DIR}/services/audio-assistant;TONESCOPE_CLI_PATH=$<TARGET_FILE:tonescope_cli>;TONESCOPE_NODE_PATH=${TONESCOPE_NODE_EXECUTABLE}")
endif()
install(TARGETS tonescope_cli RUNTIME DESTINATION bin)
if(TONESCOPE_DESKTOP)
  install(TARGETS ToneScope RUNTIME DESTINATION bin)
endif()
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "ToneScope-0.1.0-dev-win64")
include(CPack)
